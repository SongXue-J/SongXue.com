{{/* Article navigation — filtered by first category */}} {{- $reversePagination := .Scratch.Get "reversePagination" | default false -}} {{- $currentCategories := .Params.categories | default (slice) -}} {{- $firstCategory := index $currentCategories 0 | default "" -}} {{- $prev := false -}} {{- $next := false -}} {{- if $firstCategory }} {{- $allPages := where .Site.RegularPages "Params.categories" "intersect" (slice $firstCategory) -}} {{- $sortedPages := sort $allPages "Date" "desc" -}} {{- $currentIndex := -1 -}} {{- range $i, $p := $sortedPages -}} {{- if eq $p.RelPermalink $.RelPermalink -}} {{- $currentIndex = $i -}} {{- end -}} {{- end -}} {{- if ge $currentIndex 0 -}} {{- if $reversePagination -}} {{- if gt $currentIndex 0 -}} {{- $prev = index $sortedPages (sub $currentIndex 1) -}} {{- end -}} {{- if lt $currentIndex (sub (len $sortedPages) 1) -}} {{- $next = index $sortedPages (add $currentIndex 1) -}} {{- end -}} {{- else -}} {{- if lt $currentIndex (sub (len $sortedPages) 1) -}} {{- $prev = index $sortedPages (add $currentIndex 1) -}} {{- end -}} {{- if gt $currentIndex 0 -}} {{- $next = index $sortedPages (sub $currentIndex 1) -}} {{- end -}} {{- end -}} {{- end -}} {{- else -}} {{- $prev = cond $reversePagination .PrevInSection .NextInSection -}} {{- $next = cond $reversePagination .NextInSection .PrevInSection -}} {{- end -}} {{- if or $prev $next -}}
{{- if $prev -}} {{- $linkTitle := partial "utils/title" $prev -}} {{- partial "utils/icon.html" (dict "name" "chevron-right" "attributes" "class=\"hx-inline hx-h-5 hx-shrink-0 ltr:hx-rotate-180\"") -}} {{- $linkTitle -}} {{- end -}} {{- if $next -}} {{- $linkTitle := partial "utils/title" $next -}} {{- $linkTitle -}} {{- partial "utils/icon.html" (dict "name" "chevron-right" "attributes" "class=\"hx-inline hx-h-5 hx-shrink-0 rtl:-hx-rotate-180\"") -}} {{- end -}}
{{- end -}}
